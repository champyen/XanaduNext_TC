DEF("Liese")

RES("INIT")
{

	// EV 赤子を抱いた母親
	check_game_flag(3514)
	on_jump NEXT

	jump EVENT1

//--------------------------------------------------------------------

NEXT:
	jump EXIT


//-- 赤子を抱いた母親の降靈の儀式 ------------------------------------

EVENT1:
	stop_bgm(1,1000)

	ENTRY_CHR(7, 534)
	PUT_CHR( 7, "Mother1",    0.00,0.50, 2, 00)
	set_pos("Mother1",0,0.5,2)
	set_deg_ex("Mother1","Liese")
	set_deg_ex("Liese","Mother1")

	STOP_PC(0)
	set_motion("PC", "STAND",-1)
	set_deg_ex("PC","Mother1")
	look_chr("PC", "Mother1")

//　　精靈堂に入ると、祭司リーゼが降靈の儀式を行っているムービー。
//　　イベント專用ＮＰＣ(赤子を抱いた母親)が祭壇の前にかがんで
//　　いて、光に包まれている。
//　　以降、降靈の儀式イベント畫面へ。
//　　以前あったイベントを流用します。
//    同じカメラが續いてしまうので、こっちのは若干あっさりめにしてみました（井

	set_skip_label("SKIP3")

	// SE再生
	SE("SE_EVENT_00")

	set_motion("Liese","EMPTY4",0)
	set_motion("Liese2","EMPTY1",0)

	CINEMA_ON(0)
	permit_script_camera(1)
	set_script_camera(0.12,0.50,-4.56,  -45.5,225.0,0.0, 12, 15.0)
	move_script_look_at(0.0,0.00,5.0, 200, 1)
	wait(200)

	fade_out(60)
	wait(100)

	permit_script_camera(1)
	set_script_camera_ex("Liese",1, -5,200,0, 5, 40)
	scale_script_camera( 1.2, 380, 0 )
	fade_in(60)
	wait(300)

	cross_fade(40)
	SET_SCRIPT_CAMERA( 0.2,0.5,1,  -60,-30,0, 6, 30 )
	scale_script_camera( 0.8, 360, 0 )
	wait(300)

	fade_out(60)
	wait(60)
	STOP_ENV_SE("SE_EVENT_00", 2000)
	wait(120)

//　　儀式が終わると、基本畫面に戻して
//　　以下のメッセージを表示。

	set_pos("PC",1.24,0.50,-4.52)
	set_motion("Liese2","STAND",0)
	DELETE_EFC("Liese2")
	DELETE_EFC("Liese")
	set_motion("Liese","STAND",0)

	//座り老婆と插げ替え
	HIDE_CHR("Mother1")
	ENTRY_CHR(8, 122)
	PUT_CHR( 8, "Mother2",    0.00,0.50, 2, 00)
	set_pos("Mother2",0,0.5,2)
	set_deg_ex("Mother2","Liese")
	LOOK_CHR("Mother2","Liese")
	LOOK_CHR("PC","Mother2")

	// BGM 戻す
	bgm(-1, 0, -1)
	set_script_camera_EX("Mother2",1,-45,225.0,0.0, 12, 15.0)
	fade_in(60)
	wait(60)

	MSG_EX("#S3#C0莉絲「……這樣降靈的儀式就結束了。」",-1,2)
	MSG_EX("#S3#C0莉絲「峠片中的守護精靈|　　　已經順利地寄宿在孩子的身體裡了。」",-1,2)

	LOOK_CHR("Mother2","-")
	MSG_EX("#S3#C0母親「哦哦，小寶寶……太好了。」",-1,2)
	MSG_EX("#S3#C0母親「莉絲大人，非常感謝您。」",-1,2)
	set_motion("Mother2","ATTACK2",30)
	WAIT(70)

	MSG_EX("#S3#C0莉絲「不客氣……|　　　祝願孩子的身心都能健康地成長。」",-1,2)
	MSG_EX("#S3#C0莉絲「願新的生命之光受到守護精靈的保佑。」",-1,2)
	WAIT(30)

//　　イベント後、親子が退場してプレイヤーに操作許可。

	INIT_WALK_DST("Mother2")
	set_walk_dst("Mother2", 0.07,0.50,-4.40)
	walk_pos("Mother2",0,0.02)
	move_script_look_at(0.0,0.00,-5.0, 200, 1)

	wait_walk("Mother2")

	se("SE_DOOR")
	switch_on("Door1")
	wait(60)

	INIT_WALK_DST("Mother2")
	set_walk_dst("Mother2", 0.07,0.50,-7.40)
	walk_pos("Mother2",0,0.02)

	fade_out(60)
	wait(60)

SKIP3:
	// BGM 戻す
	bgm(-1, 0, -1)
	STOP_ENV_SE("SE_EVENT_00")
	HIDE_CHR("Mother1")
	HIDE_CHR("Mother2")
	LOOK_CHR("PC","-")
	set_pos("PC",1.24,0.50,-4.52)
	set_motion("Liese2","STAND",0)
	DELETE_EFC("Liese2")
	DELETE_EFC("Liese")
	set_motion("Liese","STAND",0)

	// EV 赤子を抱いた母親
	game_flag_on(3514)

	FOCUS_CHR("PC")
	permit_script_camera(0)
	CINEMA_OFF(0)
	fade_in(60)
	switch_off("Door1")
	end

EXIT:
	END

}
