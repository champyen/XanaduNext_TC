// "祈りを捧げる" ////////////////////////////////////////////////////////////
@lv_up
{
	MSG_EX("是、是的。|知道了，是祈禱吧！",-1,4)

	// 經驗值チェック --------------------------------------------------------
	check_lvup(1)
	off_jump not_enough

	MSG_EX("那個，|與祈禱一起進行降神的儀式，|可以嗎？",-1,4)
	sel_ex("　　　#c2どうしますか？|・別のガーディアンを降神する|・ガーディアンは變更しない",-1,4)
	check_select(0)
	off_jump not_change_god

	MSG_EX("我知道了。",-1,4)
	MSG_EX("那麼祈禱和降神儀式開始了。",-1,4)
	MSG_EX("請選擇守護神。",-1,4)
	
	call fo_MIKO
//	wait(30)

	call_shop(7)

	call fi_MIKO
	wait(30)
	
	check_use_shop(0)
	off_jump lvup_only
	
	MSG_EX("那麼祈禱和降神儀式開始了。",-1,4)
	call lv_up_event
	MSG_EX("似乎順利得到了新守護神的保護了呢。",-1,4)
	MSG_EX("願光明守護你和你命運的掌管者。",-1,4)
	ret

lvup_only:
	MSG_EX("那麼，就只進行祈禱。",-1,4)
	jump not_change_god2
	ret

not_change_god:
	MSG_EX("是、是嗎……",-1,4)
	MSG_EX("那麼，就只進行祈禱。",-1,4)
not_change_god2:
	call lv_up_event
	MSG_EX("似乎順利接受祝福了呢。",-1,4)
	MSG_EX("太好了。",-1,4)
	ret

not_enough:
	MSG_EX("那麼就馬上……",-1,4)
	MSG_EX("……哎？",-1,4)
	MSG_EX("那、那個……對不起。|好像是經驗值不足。",-1,4)
	ret
}

// レベルＵＰイベント --------------------------------------------------------
@lv_up_event
{
	call fo_MIKO
	set_motion("PC","EMPTY5",-1)
	set_pos("PC",0,0.5,1.4)
	set_deg_ex("PC","mag")
	set_deg_ex("mag","PC")
	wait(30)

	// 祈り演出
	cross_fade(60)

// スキップ許可
skip_enable(1)

	set_motion("mag","EMPTY1",0)
	permit_script_camera(1)
	set_script_camera_ex("mag",1, 5,200,0, 4, 40)
	scale_script_camera( 1.2, 380, 0 )
	wait(220)
	cross_fade(40)
	SET_SCRIPT_CAMERA( 0,1.8,3.3, -5,170,0, 5, 30 )
	ROTATE_SCRIPT_CAMERA( -10,-10,0, 120,0)
	wait(120)
	set_motion("mag2","EMPTY1",0)

	cross_fade(40)
	set_script_camera_ex("PC",1, -20,180,0, 7.2, 40)
	scale_script_camera( 0.9, 600, 0 )

// スキップ停止
skip_enable(0)

	set_motion("mag","EMPTY3",0)
	call_shop(6)
	set_motion("mag2","STAND",0)
	DELETE_EFC("mag2")
	set_motion("PC","EMPTY6",-1)

	cross_fade(60)
	permit_script_camera(0)

	call fi_MIKO
	wait(30)

	ret
}

// "カードを鑑定する" ////////////////////////////////////////////////////////
@analyze_card
{
	check_card(0)
	on_jump analyze
	
	MSG_EX("那、那個，對不起……|好像峠片一張也沒有了。",-1,4)
	ret
	
analyze:
	MSG_EX("那個……現在鑑定你所持有的神秘峠片。",-1,4)
	MSG_EX("可以嗎？",-1,4)
	sel_ex("よろしいですか？|#c2鑑定してもらう|やっぱりいい",-1,4)
	check_select(0)
	off_jump sel1

	MSG_EX("是、是的！|那麼我就鑑定了。",-1,4)
	call fo_MIKO

	cross_fade(60)
	set_pos("PC",0,0.5,1.4)
	set_deg_ex("PC","mag")
	set_deg_ex("mag","PC")
	set_motion("mag","EMPTY4",-1)

	permit_script_camera(1)
	SET_SCRIPT_CAMERA( 0,1.7,3.3, -20,180,0, 4, 40 )
	scale_script_camera( 1.2, 180, 0 )
	wait(180)
	call_shop(9)
	DELETE_EFC("mag")
	set_motion("mag","STAND",-1)

	cross_fade(60)
	permit_script_camera(0)
	call fi_MIKO
	wait(30)
	
	MSG_EX("怎麼了？|碰到好的守護神了嗎？",-1,4)

	ret

sel1:
	MSG_EX("是、是嗎……",-1,4)
	ret
}

	OFF_JUMP NEXT
	// --------------------------------------------------------------------------------

	// つぎのイベント用--------------------------------------------------------------
//	CHECK_GAME_FLAG(34**)
//	ON_JUMP NEXT
	// --------------------------------------------------------------------------------

	END

NEXT:
	STOP_PC(0)

	call entry_MIKO

	cinema_on(0)
	
	set_deg_ex("mag","")
	active_chr("mag")
	focus_chr("mag")
	SET_NAME("MAG","莉絲洛特")
	set_motion("mag","ATTACK1",-1)
	
	wait(120)
	call fi_MIKO
	wait(30)

	// スタート時イベント--------------------------------------------------------------
	CHECK_GAME_FLAG(3400)
	ON_JUMP NEXT
	call start_event
	NEXT:
	// --------------------------------------------------------------------------------

	set_motion("mag","ATTACK1",-1)

	call fo_MIKO
	wait(30)

	cinema_off(0)

	call erase_MIKO

	end
}
