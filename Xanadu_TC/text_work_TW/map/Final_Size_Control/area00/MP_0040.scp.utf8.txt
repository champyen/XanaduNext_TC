// 巫女畫像(512x512)讀み込み＆チップ登錄
@entry_MIKO
{

	entry_img(1,"miko_b",0)
	entry_img(2,"miko_c",0)
	entry_img(3,"miko_d",0)
	//set_img( 0, 640,  0, 0, 0,0,256,256, 0, 0x00ffffff, 0xff000000, 1,1, 0, 0,1)
	set_img( 0, 184,  0, 0, 0,0,256,256, 0, 0x00ffffff, 0xff000000, 1,1, 0, 0,1)
	img_on(0)
	//set_img( 1, 896,  0, 1, 0,0,256,256, 0, 0x00ffffff, 0xff000000, 1,1, 0, 0,1)
	set_img( 1, 440,  0, 1, 0,0,256,256, 0, 0x00ffffff, 0xff000000, 1,1, 0, 0,1)
	img_on(1)
	//set_img( 2, 640,256, 2, 0,0,256,256, 0, 0x00ffffff, 0xff000000, 1,1, 0, 0,1)
	set_img( 2, 184,256, 2, 0,0,256,256, 0, 0x00ffffff, 0xff000000, 1,1, 0, 0,1)
	img_on(2)
	//set_img( 3, 896,256, 3, 0,0,256,256, 0, 0x00ffffff, 0xff000000, 1,1, 0, 0,1)
	set_img( 3, 440,256, 3, 0,0,256,256, 0, 0x00ffffff, 0xff000000, 1,1, 0, 0,1)
	img_on(3)
	ret
}

// 巫女消去
@erase_MIKO
{
	img_off(0)
	img_off(1)
	img_off(2)
	img_off(3)
	ret
}

// ============================================================================
// コリンちゃん
// ============================================================================
DEF("mag")
RES("CLICK")
{
	call entry_MIKO
	cinema_on(0)

	set_deg_ex("mag","")
	SET_NAME("MAG","莉絲洛特")
	set_motion("mag","ATTACK1",-1)
	call fi_MIKO
	wait(30)

	check_room_flag(1)
	off_jump start
	check_room_flag(2)
	on_jump start
	
//縮圖を伸ばす----------------------------------------------------
	call skill_up_event
	set_motion("mag","ATTACK1",-1)
	call fo_MIKO
	wait(30)
	cinema_off(0)
	call erase_MIKO
	room_flag_on(2)
	end


//本編------------------------------------------------------------
start:
	check_room_flag(0)
	on_jump sel_menu
	
	MSG_EX("你好。",-1,4)
	MSG_EX("…活著的所有人，|身體中都寄宿著「靈魂」。",-1,4)
	MSG_EX("……真是不可思議的人，|竟然什麼都沒有寄宿在你的身體裡？",-1,4)
	MSG_EX("如果是你的話，也許能夠熟練操縱|留在寺院裡的秘寶……",-1,4)
	SE("SE_GETITEM")
	SYS("#C2得到靈魂之板！")
	MSG_EX("首先將中央的「開始的靈魂」|那塊石板進行降神可以吧。",-1,4)
	WAIT(30)
	call fo_MIKO
	CROSS_FADE(30)
	PERMIT_SCRIPT_CAMERA(1)
	SET_SCRIPT_CAMERA( 0,0,0, -60,-180,0, 20,32)
	SCALE_SCRIPT_CAMERA(0.9,180,0)
	skip_enable(1)
	show_surface(1)
	WAIT(180)
	call fi_MIKO
	CROSS_FADE(30)
	PERMIT_SCRIPT_CAMERA(0)
	skip_enable(0)
	MSG_EX("如果有問題的話請再來……",-1,4)

	set_motion("mag","ATTACK1",-1)
	call fo_MIKO
	wait(30)
	cinema_off(0)
	call erase_MIKO
	room_flag_on(0)
	end


//質問コーナー----------------------------------------------------
sel_menu:
	SEL("您好，有什麼事情嗎？|#C5■守護神是？|■這個房間是？|■石板的使用方法是？|■沒什麼")

	check_select(0)
	off_jump sel1
	//アニマ説明
	MSG_EX("「靈魂」是存在於這個世界的神，",-1,4)
	MSG_EX("所有的人一生下來，|身上就宿有某個方面的靈魂。",-1,4)
	jump sel_menu
	
sel1:
	check_select(1)
	off_jump sel2
	//縮圖について
	MSG_EX("展現在眼前的花紋|就是「靈魂」的縮略圖。",-1,4)
	MSG_EX("在中間的被稱為「開始的靈魂」，|是所有靈魂的起源。",-1,4)
	jump sel_menu

sel2:
	check_select(2)
	off_jump sel3
	//プレートの使い方について。
	MSG_EX("因為你的身上什麼都沒有寄宿，",-1,4)
	MSG_EX("所以可以為你寄宿多個靈魂。",-1,4)

	MSG_EX("但是，為了寄入新的靈魂，|有必要顯示現在寄宿靈魂的力量。",-1,4)
	MSG_EX("當得到足夠的熟練度之後|就會給予你那種能力（技能），",-1,4)
	MSG_EX("縮略圖中開啟了新的窓口，|可以進行新的靈魂的更換。",-1,4)
	MSG_EX("到那個時候請到這裡來哦。",-1,4)
	jump sel_menu

sel3:
	set_motion("mag","ATTACK1",-1)
	call fo_MIKO
	wait(30)
	cinema_off(0)
	call erase_MIKO
	end
}

// ============================================================================
// スキルアップイベント
// ============================================================================
@skill_up_event
{
	
	MSG_EX("太厲害了！|靈魂的力量能夠為己所用呢。",-1,4)
	MSG_EX("明白了，讓我來為你開啟新的道路吧……",-1,4)
	
	FADE_OUT_EX(30, 0,"0xff000000", 0)
	wait(30)
	call fo_MIKO
	set_motion("PC","EMPTY5",-1)
	set_pos("PC",0,0,15,1.4)
	set_deg_ex("PC","mag")
	set_deg_ex("mag","PC")
	wait(30)
	FADE_IN_EX(30, 0,"0xff000000", 0)

	// 祈り演出
	skip_enable(1)

	set_motion("mag","EMPTY1",0)
	permit_script_camera(1)
	set_script_camera_ex("mag",1, 5,200,0, 4, 40)
	scale_script_camera( 1.2, 380, 0 )
	wait(220)

	//2カメラ
	cross_fade(40)
	SET_SCRIPT_CAMERA_EX("mag",1, -5,170,0, 5, 30 )
	ROTATE_SCRIPT_CAMERA( -10,-10,0, 120,0)
	wait(120)

	//3カメラ
	cross_fade(40)
	set_script_camera_ex("PC",1, -20,180,0, 7.2, 40)
	scale_script_camera( 0.9, 600, 0 )
	set_motion("mag","EMPTY3",0)
	set_motion("PC","EMPTY6",-1)

	//縮圖アップ
	CROSS_FADE(30)
	SET_SCRIPT_CAMERA( 0,0,0, -60,-180,0, 20,32)
	SCALE_SCRIPT_CAMERA(0.9,180,0)
	WAIT(90)
	FADE_OUT_EX(5, 0,"0xffffffff", 0)
	WAIT(5)
	show_surface(2)
	FADE_IN_EX(30, 0,"0xffffffff", 0)
	WAIT(85)
	skip_enable(0)

	cross_fade(60)
	permit_script_camera(0)
	call fi_MIKO
	wait(30)
	MSG_EX("這樣的話就應該可以|更換為新的靈魂了。",-1,4)
	MSG_EX("加油啊！",-1,4)


	ret
}
