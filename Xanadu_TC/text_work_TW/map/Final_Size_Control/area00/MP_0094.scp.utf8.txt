// 靈藥を作ってもらう

@MAKE_POTION
{
	SEL("　製作靈藥嗎？　|#C2是|否")
	check_select(0)
	on_jump YES

//　　⇒「いいえ」の場合

	MSG("是這樣啊，|需要的話隨時都可以來找我。")
	ret

//　　→操作フリー
//　　再度話しかけるとメニューに[藥を作ってもらう]を表示。
//　　選擇肢に接續する。

//　　⇒「はい」の場合
YES:
	MSG("那麼，請稍等一下。")
	MSG("我也是第一次製作這種藥，|所以不查查書的話……呵呵……")
	MSG("哎呀～我已經躍躍欲試了。")

//　　畫面ブラックアウト＆フェードイン
	fade_out(60)
	wait(300)
	fade_in(60)
	wait(60)

	MSG("……讓您久等了，已經調好了哦。")

	SE("SE_GETITEM")
	SYS_EX("得到【不死鳥之靈藥】。",-1,0X00000200)
	PICK_ITEM(193)
	GET_ITEM(194, 1)
	// 3806	TK スペンス、不死鳥の靈藥を作った
	game_flag_on(3806)

	MSG("這就是從仙那度時代開始流傳的，|連不治之症都可以治好的靈藥。")
	MSG("如果有朋友生病了的話，|請一定要讓他喝下這個。")
	MSG("保證有療效哦。")

	ret
}
