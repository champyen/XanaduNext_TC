	SET_DEG("Gunter", 180)

	CINEMA_ON(0)

	STOP_PC(0)
	ACTIVE_CHR("SCREEN")
	SEL("　外出嗎？　|#C2外出|取消")

	CHECK_SELECT(0)
	ON_JUMP SELALL_1
	JUMP SELALL_2

	WAIT(30)
	SHAKE(0,10)
	SHAKE(0,10)
	SWITCH_ON("gate")
	WAIT(30)
	FADE_OUT(-1)
	GAME_FLAG_ON(3000)

	WAIT(40)
	SE("SE_CHG_AREA")
	WAIT(20)
	ENTRY_MAP("0710_out")
	END

SELALL_2:
	CINEMA_OFF(0)
	END

}
