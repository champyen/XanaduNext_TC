//////////////////////////////////////////////////////////////////////
// ドヴォルザーク

DEF("EV_Dvorak")

RES("EVENT")
{

	// EV 宿酒場のイベント(調查開始)
	check_game_flag(3502)
	off_jump EXIT
	// EV ドヴォルザークとの對決イベント
	check_game_flag(3520)
	on_jump EXIT

	check_game_flag(1024)
	off_jump EXIT
	check_game_flag(1025)
	off_jump EXIT

EVENT:
	STOP_PC(0)
	CINEMA_ON(0)

	set_skip_label("SKIP")

//　　地下遺跡の最奧まで來ると、古びた臺座の上に寶箱が２つ並ん
//　　でいる。中に入っていた「クラウン」と「いにしえの石板」を
//　　入手して引き返そうとすると、不意に入り口から聲をかけられ
//　　る。

	cross_fade(120)
	reset_map_rot_y(1,1)
	wait(2)
	permit_script_camera(1)
	set_script_camera(17.30,-8.00,-0.13,  -25.5,225.0,0.0, 12, 15.0)
	move_script_look_at(-10,0,0, 300, 1)
	rotate_script_camera(0,40,0, 600, 1)

	set_pos("PC", 16.75,-10.00, 0.0)
	set_deg("PC", 270)
	PC_WEAPON(0)
	set_motion("PC", "STAND",-1)
	show_chr("Dvorak")
	load_npc_eqp(7,00)
	dvorak(0)
	set_pos("Dvorak", 4.86,-10.00, 0.00)
	set_deg("Dvorak", 90)

	INIT_WALK_DST("PC")
	set_walk_dst("PC", 15.35,-10.00,0.0)
	walk_pos("PC",0,0.025)

	wait(60)

	MSG_EX("#S3#C0謎之劍士「……哦，閣下是騎士？」",-1,2)

	set_motion("Dvorak","STAND",-1)
	look_chr("Dvorak", "-")
	INIT_WALK_DST("Dvorak")
	set_walk_dst("Dvorak", 10.35,-10.00,0.0)
	walk_pos("Dvorak",0,0.025)

	MSG_EX("#S3#C0謎之劍士「在這樣的地方做些像盜賊一樣的勾當，|　　　　　騎士也墮落了。」",-1,2)
	wait_walk("Dvorak")
	wait_camera(0)

	cross_fade(100)
	permit_script_camera(0)
	reset_map_rot_y(7,1)
	wait(2)
	permit_script_camera(1)
	set_script_camera(10.35,-9.00,0.0,  -25.5,155.0,0.0, 12, 15.0)
	move_script_look_at(4,0,0, 300, 1)
	rotate_script_camera(0,-40,0, 500, 1)
	wait(100)

	MSG_EX("#S3#C0謎之劍士「算了……|　　　　　我想要你手中的王冠。」",-1,2)
	MSG_EX("#S3#C0謎之劍士「在它落入那個人手中之前。」",-1,2)

//　　ドヴォルザークが主人公の前まで悠然と步いて移動。
//　　主人公はやや後退。

	MSG_EX("#S3#C0謎之劍士「我不想浪費口舌。|　　　　　那不是不知緣由的你應該拿的東西。」",-1,2)

	wait(40)
	stop_camera(0)
//　　劍を拔き、身構える主人公。
	wait(20)

	set_script_camera_EX("PC",0,  -25.5,105.0,0.0, 10, 15.0)
	PC_WEAPON(1)
	move_script_look_at(-4,0,0, 200, 1)
	wait_camera(0)

	MSG_EX("#S3#C0謎之劍士「騎士不能做無謂的後退嗎……|　　　　　我並不討厭這種驕傲的精神。」",-1,2)
	MSG_EX("#S3#C0謎之劍士「也許你也有自己的理由，|　　　　　不過如果不給我的話，那就沒辦法了。」",-1,2)

//　　ドヴォルザーク、劍の鞘を払う。
	// 拔刀モーションの再生…

	se("SE_WEAPON_O")
	set_motion("Dvorak","WEAPON_O")
	// 拔刀モーション再生待ち…
	wait(54)
	// モーションテーブル讀み替え
	dvorak(1)
	// 拔刀用テーブル側のSTANDに
	set_motion("Dvorak","STAND")

	MSG_EX("#S3#C0謎之劍士「接招吧。」",-1,2)

//　　主人公とドヴォルザークが數合斬り合ったあと、距離を取る。

	INIT_WALK_DST("Dvorak")
	set_walk_dst("Dvorak", 14.0,-10.00,0.0)
	walk_pos("Dvorak",1,0.14)

	fade_out(40)

	wait(10)
	set_motion("PC","GUARD")

	wait(20)
	set_motion("PC","STAND")

//	se("SE_FOOT_G")
//	wait(8)
//	se("SE_FOOT_G")
//	wait(8)
//	se("SE_FOOT_G")
//	wait(8)

	wait(30)
	wait_walk("Dvorak")

	se("SE_GUARD")
	wait(30)
	se("SE_GUARD")
	wait(4)
	se("SE_GUARD")
	wait(10)
	se("SE_GUARD")
	wait(50)
	se("SE_DAMAGE_P")
	wait(30)
	se("SE_GUARD")
	wait(50)

	MSG_EX("#S3#C0謎之劍士「哈，相當不錯嘛……」",-1,2)

	set_script_camera_EX("Dvorak",0,  -25.5,225.0,0.0, 10, 15.0)
	rotate_script_camera(0,8,0, 300, 1)

	set_pos("Dvorak" , 10.35,-10.00,0.0)
	dvorak(0)

	fade_in(60)
	wait(80)

	MSG_EX("#S3#C0謎之劍士「害怕了嗎……|　　　　　不，應該説是迷惑了吧。」",-1,2)
	MSG_EX("#S3#C0謎之劍士「用這樣遲鈍的劍，|　　　　　是比不過我德沃夏克的。」",-1,2)

//　　ドヴォルザークが體にオーラをまとい、劍を構えなおす。

	se("SE_WEAPON_O")
	set_motion("Dvorak","WEAPON_O")
	// 拔刀モーション再生待ち…
	wait(54)
	// モーションテーブル讀み替え
	dvorak(1)
	// 拔刀用テーブル側のSTANDに
	set_motion("Dvorak","STAND")

	wait(10)

	fade_out(50)
	wait(60)

	MSG_EX("#S3#C0德沃夏克「至少能勇敢戰死，|　　　　　堅持騎士的尊嚴也不錯。」",-1,2)
	MSG_EX("#S3#C0德沃夏克「喝啊——！！」",-1,2)

//　　次の瞬間、スキル技を繰り出すドヴォルザーク。
//　　技を受けて膝から崩れ落ちる主人公。その周りは花が咲くよう
//　　に血だまりが廣がっていく。ドヴォルザーク、主人公の懐から
//　　クラウンを取り出す。

	se("SE_MAGIC")
	wait(40)

	se("SE_DAMAGE_P")
	wait(5)
	se("SE_DAMAGE_P")
	wait(5)
	se("SE_DAMAGE_P")
	wait(5)
	se("SE_DAMAGE_P")
	wait(20)
	se("SE_DAMAGE_P")
	wait(50)

	set_pos("Dvorak", 13.23,-10.00,-0.24,)
	set_deg("Dvorak", 90)

	se("SE_DEAD_P")
	set_motion("PC", "DEAD",-1)
	dvorak(0)
	// 拔刀用テーブル側のSTANDに
	set_motion("Dvorak","STAND")
	wait(10)

	fade_in(100)
	move_script_look_at(-2,0,0, 400, 1)
	wait(160)

	set_turn_spd("Dvorak", 16)
	set_deg("Dvorak", 270)

	MSG_EX("#S3#C0德沃夏克「如果不快一點的話……」",-1,2)

	fade_out(120)
	wait(200)

	// クラウンと石版を消す
//	PICK_ITEM(270)
	PICK_ITEM(187)
	pick_text_item(0)

	PC_WEAPON(0)
	wait(100)
	game_flag_on(3520)
	entry_map("0091_1009")
	permit_script_camera(0)
	CINEMA_OFF(0)

EXIT:
	END

SKIP:
	fade_out(0)
	wait(1)

	// クラウンと石版を消す
//	PICK_ITEM(270)
	PICK_ITEM(187)
	pick_text_item(0)

	PC_WEAPON(0)
	wait(60)
	game_flag_on(3520)
	entry_map("0091_1009")

	END
}
