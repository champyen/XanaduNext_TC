RES("WIN")
{
	CHECK_GAME_FLAG(1032)
	ON_JUMP next

	STOP_PC(0)
	CINEMA_ON(0)
	
	RESET_MAP_ROT_Y(6,90)
	FOCUS_CHR("null")
	WAIT(90)
	SHOW_CHR("trs00")
	SHOW_CHR("trs01")
	EFC("EFC_ADD_TREASURE",9.25,0,3.7)
	EFC("EFC_ADD_TREASURE",9.25,0,-3.7)
	WAIT(120)

	GAME_FLAG_ON(3526)
	GAME_FLAG_ON(1032)

	//----------------------------
	novel_on(0,1)
	wait(60)
	SYS_EX("關於【地圖的轉動】",-1,0X00000200)
	SYS_EX("在大多數迷宮中，|可以透過旋轉視角來更好地看清周圍。",-1,0X00000200)
	SYS_EX("轉動滑鼠的滾輪/點撃手把旋轉視角鍵，|可以旋轉地圖的方向。",-1,0X00000200)
	SYS_EX("將地圖轉至易看的角度，|有可能會發現隱藏的寶箱。",-1,0X00000200)
	SYS_EX("但是，也有一些地圖是不能轉動的。",-1,0X00000200)
	novel_off(0)
	wait(60)
	//----------------------------

	FOCUS_CHR("PC")

	CINEMA_OFF(0)

next:
}

DEF("trs00")
RES("INIT")
{
	CHECK_GAME_FLAG(1029)
	OFF_JUMP next

	INIT_SWITCH_ON("trs00")

next:
}

RES("CLICK")
{
	CINEMA_ON(0)
	STOP_PC(0)
	FOCUS_CHR("trs00")

// 20ゴールド
	SWITCH_ON("trs00")
	POP_ITEM_ICON("trs00", 225, 9.25,0,-3.7, 1,80,0)
	WAIT(140)

	DEL_ITEM_ICON("trs00")
	ATTACH_EFC("EFC_99ITEM_GET","trs00","POS")

	SYS_EX("得到【20Ｇ】！",-1,0X00000200)

	GET_GOLD(20)
	GAME_FLAG_ON(1029)
	FOCUS_CHR("PC")
	CINEMA_OFF(0)
	END

EXIT:
}
